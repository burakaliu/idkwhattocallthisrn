import React, { useState, useEffect } from "react";
import { invoke } from "@tauri-apps/api/core";
import { Save } from "lucide-react";

// Define our settings interface
interface Settings {
  theme: {
    primary: string;
    secondary: string;
    background: string;
  };
  notifications: {
    enabled: boolean;
    sound: boolean;
  };
  timerDefaults: {
    workDuration: number;
    breakDuration: number;
  };
}

const SettingsPage = () => {
  // Initialize with default settings
  const [settings, setSettings] = useState<Settings>({
    theme: {
      primary: "#3B82F6",
      secondary: "#10B981",
      background: "#FFFFFF",
    },
    notifications: {
      enabled: true,
      sound: true,
    },
    timerDefaults: {
      workDuration: 25,
      breakDuration: 5,
    },
  });

  const [saveStatus, setSaveStatus] = useState<"idle" | "saving" | "saved">(
    "idle"
  );

  // Load settings on component mount
  useEffect(() => {
    const loadSettings = async () => {
      try {
        // Invoke your Rust function to load settings
        const savedSettings = await invoke("load_settings");
        if (savedSettings) {
          setSettings(savedSettings as Settings);
        }
      } catch (error) {
        console.error("Failed to load settings:", error);
      }
    };
    loadSettings();
  }, []);

  const handleSave = async () => {
    setSaveStatus("saving");
    try {
      // Invoke your Rust function to save settings
      await invoke("save_settings", { settings });
      setSaveStatus("saved");
      setTimeout(() => setSaveStatus("idle"), 2000);
    } catch (error) {
      console.error("Failed to save settings:", error);
      setSaveStatus("idle");
    }
  };

  const handleColorChange = (key: keyof Settings["theme"], value: string) => {
    setSettings((prev) => ({
      ...prev,
      theme: {
        ...prev.theme,
        [key]: value,
      },
    }));
  };

  // Inside your Settings component, add this function:
  const testTauri = async () => {
    try {
      // Save some test settings
      await invoke("save_settings", {
        settings: {
          theme: {
            primary: "#3B82F6",
            secondary: "#10B981",
            background: "#FFFFFF",
          },
          notifications: {
            enabled: true,
            sound: true,
          },
          timerDefaults: {
            workDuration: 25,
            breakDuration: 5,
          },
        },
      });
      console.log("Settings saved successfully");

      // Try to load the settings
      const loadedSettings = await invoke("load_settings");
      console.log("Loaded settings:", loadedSettings);
    } catch (error) {
      console.error("Error testing Tauri:", error);
    }
  };

  return (
    <div className="max-w-4xl mx-auto p-6 space-y-8">
      {/* Header */}
      <div className="flex justify-between items-center">
        <h1 className="text-2xl font-bold">Settings</h1>
        <button
          onClick={handleSave}
          className="flex items-center gap-2 px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors"
        >
          <Save size={20} />
          {saveStatus === "saving"
            ? "Saving..."
            : saveStatus === "saved"
            ? "Saved!"
            : "Save Changes"}
        </button>
      </div>

      {/* Theme Settings */}
      <section className="space-y-4">
        <h2 className="text-xl font-semibold">Theme</h2>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div className="space-y-2">
            <label className="block text-sm font-medium">Primary Color</label>
            <div className="flex items-center gap-2">
              <input
                type="color"
                value={settings.theme.primary}
                onChange={(e) => handleColorChange("primary", e.target.value)}
                className="h-10 w-20"
              />
              <input
                type="text"
                value={settings.theme.primary}
                onChange={(e) => handleColorChange("primary", e.target.value)}
                className="px-3 py-2 border rounded-lg"
              />
            </div>
          </div>

          <div className="space-y-2">
            <label className="block text-sm font-medium">Secondary Color</label>
            <div className="flex items-center gap-2">
              <input
                type="color"
                value={settings.theme.secondary}
                onChange={(e) => handleColorChange("secondary", e.target.value)}
                className="h-10 w-20"
              />
              <input
                type="text"
                value={settings.theme.secondary}
                onChange={(e) => handleColorChange("secondary", e.target.value)}
                className="px-3 py-2 border rounded-lg"
              />
            </div>
          </div>
        </div>
      </section>

      {/* Notification Settings */}
      <section className="space-y-4">
        <h2 className="text-xl font-semibold">Notifications</h2>
        <div className="space-y-3">
          <label className="flex items-center gap-2">
            <input
              type="checkbox"
              checked={settings.notifications.enabled}
              onChange={(e) =>
                setSettings((prev) => ({
                  ...prev,
                  notifications: {
                    ...prev.notifications,
                    enabled: e.target.checked,
                  },
                }))
              }
              className="h-4 w-4 rounded"
            />
            <span>Enable Notifications</span>
          </label>

          <label className="flex items-center gap-2">
            <input
              type="checkbox"
              checked={settings.notifications.sound}
              onChange={(e) =>
                setSettings((prev) => ({
                  ...prev,
                  notifications: {
                    ...prev.notifications,
                    sound: e.target.checked,
                  },
                }))
              }
              className="h-4 w-4 rounded"
            />
            <span>Enable Sound</span>
          </label>
        </div>
      </section>

      {/* Timer Defaults */}
      <section className="space-y-4">
        <h2 className="text-xl font-semibold">Timer Defaults</h2>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div className="space-y-2">
            <label className="block text-sm font-medium">
              Work Duration (minutes)
            </label>
            <input
              type="number"
              value={settings.timerDefaults.workDuration}
              onChange={(e) =>
                setSettings((prev) => ({
                  ...prev,
                  timerDefaults: {
                    ...prev.timerDefaults,
                    workDuration: parseInt(e.target.value),
                  },
                }))
              }
              className="px-3 py-2 border rounded-lg w-full"
              min="1"
              max="60"
            />
          </div>

          <div className="space-y-2">
            <label className="block text-sm font-medium">
              Break Duration (minutes)
            </label>
            <input
              type="number"
              value={settings.timerDefaults.breakDuration}
              onChange={(e) =>
                setSettings((prev) => ({
                  ...prev,
                  timerDefaults: {
                    ...prev.timerDefaults,
                    breakDuration: parseInt(e.target.value),
                  },
                }))
              }
              className="px-3 py-2 border rounded-lg w-full"
              min="1"
              max="30"
            />
          </div>
        </div>
      </section>
      <button onClick={testTauri}>Test Tauri Integration</button>

    </div>
  );
};

export default SettingsPage;
